DSL 完整语法定义

本节定义 KD-DSL（Knowledge Domain DSL）的完整文法。KD-DSL 采用 JSON 作为序列化格式，支持在 PostgreSQL JSONB 中直接存储和索引。规则以 JSON 表达，求值引擎将 JSON 解析为 AST 后执行。

1.1 EBNF 文法

(* KD-DSL EBNF Grammar *)

(* 顶层规则 *)
Rule         = "{" metadata "," condition "," conclusion "}";

(* 元数据 *)
metadata     = "\"metadata\"" ":" "{"
                rule_id ","
                version ","
                category ","
                author ","
                created_at
              "}";

(* 条件表达式 — 核心 *)
condition    = atomic_condition | compound_condition;

compound_condition = and_condition
                    | or_condition
                    | not_condition
                    | any_condition;  (* 至少一个匹配 *)

and_condition = "{" "\"type\"" ":" "\"AND\"" ","
                "\"conditions\"" ":" "[" condition "," condition "}" "]" "}";

or_condition  = "{" "\"type\"" ":" "\"OR\"" ","
                "\"conditions\"" ":" "[" condition "," condition "}" "]" "}";

not_condition = "{" "\"type\"" ":" "\"NOT\"" ","
                "\"condition\"" ":" condition "}";

(* 原子条件 *)
atomic_condition = "{" "\"type\"" ":" "\"ATOMIC\"" ","
                    "\"field\"" ":" field_path ","
                    "\"operator\"" ":" operator ","
                    "\"value\"" ":" value
                    "}";

(* 字段路径：支持嵌套引用 *)
field_path   = "\"odds.macau.current.handicap\""
              | "\"odds.bet365.initial.upper\""
              | "\"match.league\""
              | "\"match.home_team\""
              | "\"features.volatility\""
              | "\"features.kelly_index\""
              | "\"features.volume_ratio\"";

(* 算子全集 *)
operator     = "\"eq\""     | // 等于
                "\"neq\""    | // 不等于
                "\"gt\""     | // 大于
                "\"gte\""    | // 大于等于
                "\"lt\""     | // 小于
                "\"lte\""    | // 小于等于
                "\"between\"" | // 在区间内 [min, max]
                "\"in\""     | // 在集合中
                "\"not_in\"" | // 不在集合中
                "\"contains\"" | // 字符串包含
                "\"regex\""  | // 正则匹配
                "\"changed\"" | // 值是否变化（初盘→当前）
                "\"changed_by\""; // 变化量绝对值或百分比

(* 时间表达式 *)
time_condition = "{" "\"type\"" ":" "\"TIME\"" ","
                  "\"anchor\"" ":" time_anchor ","
                  "\"window\"" ":" time_window "}";

time_anchor  = "\"kickoff\""     | // 开赛时间
                "\"analysis\""   | // 分析时间
                "\"now\"";        // 当前时间

time_window  = "{" "\"before\"" ":" duration ","
                "\"after\"" ":" duration "}"
              | "{" "\"before\"" ":" duration "}"
              | "{" "\"after\"" ":" duration "}";

duration     = "\"30m\"" | "\"2h\"" | "\"1d\"" | "\"15m\"";

DSL 规约要点

DSL 规则以 JSON 格式存储于 PostgreSQL JSONB 列，支持 jsonb_path_exists 进行索引加速。规则条件采用表达式树结构，求值引擎递归遍历。所有时间表达式使用 UTC 时间戳，避免时区歧义。

2.2 qd_rule_versions — 规则版本表

CREATE TABLE qd_rule_versions (
    rule_id           VARCHAR(32)    NOT NULL,
    version           INT            NOT NULL,

    -- 规则标识
    category          VARCHAR(32)    NOT NULL
                        CHECK (category IN (
                            'odds_change',
                            'institution_diff',
                            'sensitivity',
                            'league_feature'
                        )),
    title             VARCHAR(128)   NOT NULL,
    description       TEXT,

    -- DSL 规则体
    dsl_body          JSONB          NOT NULL,
    dsl_hash          VARCHAR(64)    NOT NULL,

    -- 结论
    direction         VARCHAR(20)    NOT NULL
                        CHECK (direction IN (
                            'favor_upper', 'favor_lower',
                            'favor_home', 'favor_away',
                            'under', 'over',
                            'reversal', 'follow',
                            'caution', 'warning',
                            'neutral'
                        )),
    base_confidence   NUMERIC(4,3)   NOT NULL DEFAULT 0.500
                        CHECK (base_confidence BETWEEN 0 AND 1),

    -- 状态机
    status            VARCHAR(20)    NOT NULL DEFAULT 'draft'
                        CHECK (status IN (
                            'draft', 'proposed', 'experiment',
                            'validated', 'approved', 'active',
                            'superseded', 'deprecated'
                        )),

    -- 元数据
    author            VARCHAR(64)    NOT NULL,
    approved_by       VARCHAR(64),
    source            VARCHAR(128),
    tags              VARCHAR(32)[]  NOT NULL DEFAULT '{}',
    related_rules     VARCHAR(32)[]  NOT NULL DEFAULT '{}',

    -- 统计
    evidence_count    INT            NOT NULL DEFAULT 0,
    hit_count         INT            NOT NULL DEFAULT 0,
    accuracy          NUMERIC(4,3),

    -- 时间审计
    created_at        TIMESTAMPTZ    NOT NULL DEFAULT NOW(),
    updated_at        TIMESTAMPTZ    NOT NULL DEFAULT NOW(),

    PRIMARY KEY (rule_id, version),

    -- 同一 rule_id 下版本号递增
    CONSTRAINT ck_version_positive CHECK (version > 0)
);

-- 查询活跃规则
CREATE INDEX idx_rules_active ON qd_rule_versions (rule_id)
    WHERE status = 'active';

-- 按分类查询
CREATE INDEX idx_rules_category ON qd_rule_versions (category, status);

-- DSL 内容哈希去重
CREATE UNIQUE INDEX idx_rules_dsl_hash ON qd_rule_versions (dsl_hash);

2.5 qd_rule_conflicts — 规则冲突注册表

CREATE TABLE qd_rule_conflicts (
    conflict_id       UUID           PRIMARY KEY DEFAULT gen_random_uuid(),
    rule_id_a         VARCHAR(32)    NOT NULL,
    rule_id_b         VARCHAR(32)    NOT NULL,

    -- 冲突类型
    conflict_type     VARCHAR(20)    NOT NULL
                        CHECK (conflict_type IN (
                            'opposite_direction',  -- 方向相反
                            'partial_overlap',     -- 条件部分重叠
                            'same_condition'        -- 条件相同结论不同
                        )),

    -- 仲裁方案
    resolution        VARCHAR(20)    NOT NULL
                        CHECK (resolution IN (
                            'confidence_priority',
                            'recent_priority',
                            'manual_intervention',
                            'both_uncertain'
                        )),

    registered_by     VARCHAR(64)    NOT NULL,
    registered_at     TIMESTAMPTZ    NOT NULL DEFAULT NOW(),

    UNIQUE (rule_id_a, rule_id_b),
    CHECK (rule_id_a < rule_id_b)
);

CREATE INDEX idx_conflicts_rule ON qd_rule_conflicts (rule_id_a);
CREATE INDEX idx_conflicts_rule2 ON qd_rule_conflicts (rule_id_b);

API 契约规范

3.1 API 版本化策略

所有 API 端点以 /api/v1/ 为前缀。版本号嵌入 URL 路径，支持通过 Nginx 进行版本路由。当引入不兼容变更时，递增主版本号；向后兼容的变更保持当前版本。

遵循 RFC 7807 Problem Details 标准：

{
  "type": "https://api.kb-system.com/errors/rule-not-found",
  "title": "规则未找到",
  "status": 404,
  "detail": "rule_id=R001_v3 不存在",
  "instance": "/api/v1/rules/R001_v3",

}

4.3 瓶颈分析

DSL 求值引擎

类型: CPU 密集型（条件求值、模式匹配）

瓶颈: 规则数量 × 规则复杂度呈线性增长

优化策略:

  • 规则条件预编译为求值函数
  • 常见条件缓存求值结果
  • 规则按活跃度分区，冷规则不加载

PostgreSQL 查询

类型: IO 密集型（特征加载、证据写入）

瓶颈: 连接池耗尽、索引缺失

优化策略:

  • 连接池上限 150
  • 部分索引 + 覆盖索引
  • 证据写入批量提交

规则缓存访问

类型: 网络延迟（Redis 跨进程访问）

瓶颈: 缓存未命中时的回源查询

优化策略:

  • Worker 本地缓存（LRU, 100 条）
  • 缓存预热（启动时加载全部活跃规则）
  • 主动失效（规则变更时广播失效通知）

回测作业

类型: CPU + IO 混合

瓶颈: 大量历史数据扫描

优化策略:

  • 回测数据预计算物化视图
  • 有界 Worker 池（4 并发）
  • 与实时检索 Worker 资源隔离

融合决策层

融合决策层将 DSL 引擎输出的多条规则匹配结果、冲突组和推理链，融合为单一预测结论。这是解决"多条规则给出不同方向"问题的核心组件。

5.1 融合策略

支持三种融合策略，可通过配置动态切换：

策略算法适用场景
加权投票按规则置信度加权，方向得票最高者获胜规则数量多、置信度分布均匀
置信度优先取最高置信度规则的方向存在高置信度权威规则
贝叶斯融合以规则历史准确率为先验，计算后验概率规则历史数据充足、需概率输出

5.2 加权投票算法

interface FusionContext {
  exact_matches: RuleMatch[];
  fuzzy_matches: FuzzyMatch[];
  conflicts: ConflictGroup[];
  strategy: 'weighted_vote' | 'confidence_priority' | 'bayesian';
}

function fusePredictions(
  ctx: FusionContext
): FusionResult {

  switch (ctx.strategy) {
    case 'weighted_vote': {
      // 按方向聚合加权分数
      const votes: Record<string, number> = {};
      const allMatches = [...ctx.exact_matches, ...ctx.fuzzy_matches];

      for (const match of allMatches) {
        const weight = match.confidence * match.base_weight;
        votes[match.direction] = (votes[match.direction] || 0) + weight;
      }

      // 找到最高得分的方向
      const [bestDirection, bestScore] = Object.entries(votes)
        .sort(([, a], [, b]) => b - a)[0];

      // 计算置信度：最高分 / 总分
      const totalScore = Object.values(votes).reduce((a, b) => a + b, 0);
      const confidence = totalScore > 0 ? bestScore / totalScore : 0;

      return {
        direction: bestDirection,
        confidence: Number(confidence.toFixed(3)),
        vote_details: votes,
        strategy_used: 'weighted_vote'
      };
    }

    case 'confidence_priority': {
      // 取最高置信度规则的方向
      const best = allMatches.reduce((best, m) =>
        m.confidence > best.confidence ? m : best
      );
      return {
        direction: best.direction,
        confidence: best.confidence,
        strategy_used: 'confidence_priority'
      };
    }
  }
}

interface FusionResult {
  direction: PredictionDirection;
  confidence: number;
  vote_details?: Record<string, number>;
  strategy_used: string;
}

6.2 仲裁优先级

// 冲突仲裁器 — 按优先级链裁决
function arbitrate(
  conflict: ConflictGroup,
  context: ArbitrationContext
): ArbitrationResult {

  // 优先级 1: 人工注册的冲突仲裁方案
  const registered = await lookupRegisteredConflict(
    conflict.rule_ids
  );
  if (registered) {
    return applyResolution(registered.resolution, conflict);
  }

  // 优先级 2: 置信度优先
  const sorted = conflict.rules.sort(
    (a, b) => b.confidence - a.confidence
  );
  const gap = sorted[0].confidence - sorted[1].confidence;

  if (gap >= 0.15) {
    // 置信度差距 ≥ 0.15，取高置信度规则
    return { winner: sorted[0], reason: 'confidence_gap' };
  }

  // 优先级 3: 证据量优先
  const byEvidence = sorted.sort(
    (a, b) => b.evidence_count - a.evidence_count
  );
  if (byEvidence[0].evidence_count >= byEvidence[1].evidence_count * 2) {
    return { winner: byEvidence[0], reason: 'evidence_dominance' };
  }

  // 优先级 4: 最近更新优先
  const byRecent = sorted.sort(
    (a, b) => b.updated_at - a.updated_at
  );
  return { winner: byRecent[0], reason: 'recent_update' };
}

6.3 冲突仲裁结果影响

仲裁结果影响融合决策层的输入：

  • 仲裁失败的规则（未获胜方）在融合决策中权重降低 50%
  • 仲裁结果为 both_uncertain 时，融合决策层输出 caution 方向
  • 仲裁结果为 manual_intervention 时，系统发出告警，等待人工处理

7.1 时间戳三元素校验

// 核心不变量：observed_at ≤ received_at ≤ match_time
// 同时必须满足：analysis_time ≤ match_time（分析不能在比赛开始后）

interface TemporalContext {
  // 分析的时间锚点（回测时 = 历史时间点，实时 = NOW）
  analysis_time: Date;

  // 允许的数据时间窗口
  eligible_window: {
    start: Date;  // 最早可用的数据时间
    end: Date;    // 最晚可用的数据时间 (= analysis_time)
  };

  // 严格模式：为 true 时，任何时间校验失败都拒绝分析
  // 宽松模式：为 false 时，记录警告但继续
  strict_mode: boolean;
}

function validateTemporalIntegrity(
  evidence: EvidenceInput,
  context: TemporalContext
): TemporalCheck {

  const checks: TemporalCheckItem[] = [];

  // 检查 1: observed_at ≤ received_at
  checks.push({
    name: 'observation_before_reception',
    passed: evidence.observed_at <= evidence.received_at,
    detail: `observed=${evidence.observed_at} ≤ received=${evidence.received_at}`
  });

  // 检查 2: received_at ≤ match_time
  checks.push({
    name: 'reception_before_match',
    passed: evidence.received_at <= evidence.match_time,
    detail: `received=${evidence.received_at} ≤ match=${evidence.match_time}`
  });

  // 检查 3: 数据时间 ≤ 分析时间（禁止使用未来数据）
  checks.push({
    name: 'no_future_data',
    passed: evidence.received_at <= context.analysis_time,
    detail: `data=${evidence.received_at} ≤ analysis=${context.analysis_time}`
  });

  // 检查 4: 分析时间 ≤ 比赛时间（不能在比赛开始后分析）
  checks.push({
    name: 'analysis_before_match',
    passed: context.analysis_time <= evidence.match_time,
    detail: `analysis=${context.analysis_time} ≤ match=${evidence.match_time}`
  });

  const allPassed = checks.every(c => c.passed);

  return {
    valid: allPassed,
    checks: checks,
    action: allPassed ? 'proceed'
         : context.strict_mode ? 'reject'
         : 'warn'
  };
}

7.3 反盲区防护（Look-Ahead Bias）

// 批量回测时，必须确保第 N 场比赛不会用到第 N+1 场比赛的数据
async function runBacktestWithoutLookAhead(
  matches: MatchFeatures[],
  config: BacktestConfig
): Promise<AnalysisResult[]> {
  // 1. 按比赛时间升序排列
  const sorted = [...matches].sort(
    (a, b) => a.match_time - b.match_time
  );

  const results: AnalysisResult[] = [];

  for (let i = 0; i < sorted.length; i++) {
    const match = sorted[i];

    // 2. 每个 match 的 analysis_time = match.match_time - 1h
    //    确保只使用该时间点之前的数据
    const context: TemporalContext = {
      analysis_time: new Date(
        match.match_time.getTime() - 3600000
      ),
      eligible_window: {
        start: config.date_range.start,
        end: new Date(match.match_time.getTime() - 3600000)
      },
      strict_mode: true
    };

    const result = await dsEngine.analyzeMatch(
      match.match_id, match.features, context
    );
    results.push(result);
  }

  return results;
}

async function checkDataSourceHealth(
  sourceId: string
): Promise<SourceHealthStatus> {
  const source = await pg.query(
    `SELECT * FROM qd_data_sources WHERE source_id = $1`,
    [sourceId]
  );

  const healthCheckUrl = source.health_check_url;
  if (!healthCheckUrl) {
    return { status: 'unknown', reason: 'no_health_check_url' };
  }

  try {
    const response = await fetch(healthCheckUrl, {
      timeout: 5000,
      method: 'HEAD'
    });

    const status = response.ok ? 'healthy' : 'degraded';

    // 更新数据源状态
    await pg.query(
      `UPDATE qd_data_sources
       SET last_sync_at = NOW(), last_sync_status = $1
       WHERE source_id = $2`,
      [status, sourceId]
    );

    return { status: status };
  } catch (err) {
    await pg.query(
      `UPDATE qd_data_sources
       SET last_sync_status = 'failed'
       WHERE source_id = $1`,
      [sourceId]
    );

    return { status: 'down', reason: err.message };
  }
}

9.2 跨域依赖原则

原则 1：写时复制

正在被检索 Worker 使用的规则，在置信度更新时不会被直接修改。新置信度只影响新分析任务，不影响正在进行的分析。

实现：规则缓存采用"加载时快照"模式，Worker 在一次分析中使用的规则版本在分析开始时固定。

原则 2：缓存失效优先

调度 Worker 完成回测后，不直接更新检索 Worker 的运行时状态。而是通过缓存失效通知触发热加载。

实现：Redis PUB/SUB 通道 rule:status_changed，Worker 收到后异步重新加载受影响规则。

原则 3：异步解耦

回测→置信度更新→规则状态重新评估，这三个步骤采用异步事件驱动，不阻塞任何链路。

实现：回测完成后向 qd_analysis_commands 写入一条"置信度更新"命令，由调度 Worker 领取后执行。

原则 4：最终一致性

规则置信度更新与实时分析之间允许短暂不一致（≤ 30 秒），但保证最终一致。

实现：缓存的 TTL 设为 30 秒，过期后自动重新加载。

11.3 缓存预热流程

// 检索 Worker 启动时预热
async function warmCache(workerId: string): Promise<void> {
  // 1. 从 Redis 加载所有活跃规则
  const cachedRules = await redis.get('rules:active:all');

  if (cachedRules) {
    // 缓存命中，直接加载到本地 L1 缓存
    loadToLocalCache(JSON.parse(cachedRules));
    return;
  }

  // 2. 缓存未命中，从数据库加载
  const activeRules = await pg.query(
    `SELECT * FROM qd_rule_versions WHERE status = 'active'`
  );

  // 3. 写入 Redis 缓存
  await redis.set('rules:active:all', JSON.stringify(activeRules), {
    EX: 300  // 5 分钟 TTL
  });

  // 4. 加载到本地 L1 缓存
  loadToLocalCache(activeRules);
}
