3. 容错策略

3.1 Circuit Breaker 模式

// Circuit Breaker 配置（每个依赖独立配置）
const CIRCUIT_BREAKERS = {
  // Redis 缓存断路器
  redis_cache: {
    failure_threshold: 5,          // 连续失败次数
    failure_window_ms: 30000,     // 30 秒窗口
    open_timeout_ms: 30000,       // 30 秒后进入 half-open
    fallback: 'direct_to_db'     // 降级：直接查询数据库
  },

  // PostgreSQL 断路器
  postgresql: {
    failure_threshold: 3,
    failure_window_ms: 10000,
    open_timeout_ms: 15000,
    fallback: 'reject_request'   // 拒绝请求，返回 503
  },

  // 外部数据源断路器
  external_data_source: {
    failure_threshold: 3,
    failure_window_ms: 60000,
    open_timeout_ms: 60000,
    fallback: 'use_cached_data'  // 使用缓存数据
  }
};

// 断路器实现核心逻辑
async function circuitBreakerCall<T>(
  name: string,
  fn: () => Promise<T>,
  fallback: () => Promise<T>
): Promise<T> {
  const state = await redis.get(`cb:${name}:state`);

  if (state === 'open') {
    return fallback();  // 快速失败，走降级路径
  }

  try {
    const result = await fn();
    await redis.del(`cb:${name}:failures`);
    return result;
  } catch (err) {
    const failures = await redis.incr(`cb:${name}:failures`);
    await redis.expire(`cb:${name}:failures`, 30);

    if (failures >= CIRCUIT_BREAKERS[name].failure_threshold) {
      await redis.set(`cb:${name}:state`, 'open', { EX: 30 });
    }

    return fallback();
  }
}

3.4 重试风暴防护

// 指数退避 + 抖动
async function retryWithBackoff<T>(
  fn: () => Promise<T>,
  options: {
    max_retries: number;
    base_delay_ms: number;
    max_delay_ms: number;
  } = { max_retries: 3, base_delay_ms: 1000, max_delay_ms: 30000 }
): Promise<T> {
  let lastError: Error;

  for (let attempt = 0; attempt <= options.max_retries; attempt++) {
    try {
      return await fn();
    } catch (err) {
      lastError = err;

      // 不可重试的错误，直接抛出
      if (err.status_code === 400 || err.status_code === 409) {
        throw err;
      }

      if (attempt === options.max_retries) break;

      // 指数退避 + 随机抖动
      const delay = Math.min(
        options.base_delay_ms * Math.pow(2, attempt),
        options.max_delay_ms
      );
      const jitter = delay * 0.1 * Math.random();
      await sleep(delay + jitter);
    }
  }

  throw lastError;
}

// 重试风暴防护：全局级联重试计数器
// 如果同一时间段内系统级重试次数超过阈值，熔断所有重试
const RETRY_STORM_PROTECTION = {
  global_retry_threshold: 100,  // 每分钟
  cooldown_ms: 30000           // 冷却 30 秒
};

4.2 动态配置机制

// 动态配置存储（Redis Hash）
// key: config:dynamic
// fields: 配置项名称 → 值

const DYNAMIC_CONFIG = {
  prefix: 'config:dynamic',

  // 默认值（Redis 中不存在时使用）
  defaults: {
    'lease_ttl_seconds': 30,
    'heartbeat_interval_seconds': 10,
    'cache_ttl_seconds': 300,
    'analysis_timeout_ms': 5000,
    'fuzzy_match_threshold': 0.65,
    'max_retrieval_workers': 20,
    'cb_failure_threshold': 5,
    'cb_open_timeout_ms': 30000
  },

  // 获取配置值（优先读取 Redis，fallback 到默认值）
  async get(key: string): Promise<any> {
    const value = await redis.hget('config:dynamic', key);
    return value !== null ? JSON.parse(value) : this.defaults[key];
  },

  // 更新配置值（记录审计日志）
  async set(key: string, value: any, operator: string): Promise<void> {
    const oldValue = await this.get(key);

    await redis.hset('config:dynamic', key, JSON.stringify(value));

    // 记录审计日志
    await pg.query(
      `INSERT INTO qd_audit_log (actor, action, resource_type, resource_id, changes)
       VALUES ($1, 'config.update', 'config', $2, $3)`,
      [operator, key, { before: { [key]: oldValue }, after: { [key]: value } }]
    );
  }
};

// Feature Flag 定义
const FEATURE_FLAGS = {
  // 新 DSL 求值引擎（灰度发布）
  'dsl_v2_engine': {
    description: '启用新版 DSL 求值引擎',
    owner: 'engine-team',
    created_at: '2026-08-01',
    state: 'beta',  // off | beta | stable | deprecated
    rollout_percentage: 10,  // 10% 流量
    target_users: ['internal_test']
  },

  // 贝叶斯融合策略
  'bayesian_fusion': {
    description: '启用贝叶斯融合策略替代加权投票',
    owner: 'ds-team',
    created_at: '2026-08-10',
    state: 'off',
    rollout_percentage: 0
  }
};

// Feature Flag 检查
function isFeatureEnabled(
  flagName: string,
  context?: { user_id?: string }
): boolean {
  const flag = FEATURE_FLAGS[flagName];
  if (!flag) return false;
  if (flag.state === 'off') return false;
  if (flag.state === 'stable') return true;

  // Beta 状态：按百分比或用户白名单
  if (context?.user_id && flag.target_users?.includes(context.user_id)) {
    return true;
  }

  return Math.random() * 100 < (flag.rollout_percentage || 0);
}

7.2 回滚流程

// 回滚 API 实现
async function rollbackRule(
  ruleId: string,
  targetVersion: number,
  operator: string
): Promise<RollbackResult> {
  // 1. 获取目标版本
  const target = await pg.query(
    `SELECT * FROM qd_rule_versions WHERE rule_id=$1 AND version=$2`,
    [ruleId, targetVersion]
  );

  if (target.rows.length === 0) {
    throw new Error('目标版本不存在');
  }

  // 2. 获取当前最大版本号
  const maxVersion = await pg.query(
    `SELECT MAX(version) as max_v FROM qd_rule_versions WHERE rule_id=$1`,
    [ruleId]
  );
  const newVersion = (maxVersion.rows[0].max_v || 0) + 1;

  // 3. 创建新版本（克隆目标版本的 DSL 内容）
  const newRule = await pg.query(
    `INSERT INTO qd_rule_versions
     (rule_id, version, category, title, dsl_body, dsl_hash, direction, base_confidence,
      status, author, source, tags, related_rules, description)
     SELECT $1, $2, category, title, dsl_body, dsl_hash, direction, base_confidence,
            'proposed', $3, CONCAT(source, ' (rollback from v', $4, ')'), tags, related_rules, description
     FROM qd_rule_versions
     WHERE rule_id=$5 AND version=$6
     RETURNING *`,
    [ruleId, newVersion, operator, targetVersion, ruleId, targetVersion]
  );

  // 4. 标记当前活跃版本为 superseded
  await pg.query(
    `UPDATE qd_rule_versions SET status='superseded'
     WHERE rule_id=$1 AND status='active'`,
    [ruleId]
  );

  // 5. 失效缓存
  await redis.del(`rules:${ruleId}`);

  // 6. 记录审计日志
  await pg.query(
    `INSERT INTO qd_audit_log (actor, action, resource_type, resource_id, changes)
     VALUES ($1, 'rule.rollback', 'rule', $2, $3)`,
    [operator, ruleId, { from_version: targetVersion, new_version: newVersion }]
  );

  return {
    rule_id: ruleId,
    new_version: newVersion,
    source_version: targetVersion,
    status: 'proposed',
    message: `回滚到版本 ${targetVersion} 成功，新版本 ${newVersion} 需要审核后发布`
  };
}

8.2 实现代码

class GracefulShutdown {
  private isShuttingDown = false;
  private inFlightTasks = new Set<string>();
  private shutdownTimeoutMs = 30000;
  private gracefulPeriodMs = 5000;  // 5 秒宽限期

  constructor(workerId: string, processType: string) {
    this.workerId = workerId;
    this.processType = processType;

    // 注册信号处理
    process.on('SIGTERM', async () => {
      await this.shutdown();
    });
    process.on('SIGINT', async () => {
      await this.shutdown();
    });
  }

  async shutdown(): Promise<void> {
    if (this.isShuttingDown) return;
    this.isShuttingDown = true;

    const logger = getLogger(this.workerId);
    logger.info('开始优雅关闭...', { in_flight: this.inFlightTasks.size });

    try {
      // Step 1: 停止认领新任务（标记 draining）
      await pg.query(
        `UPDATE qd_worker_heartbeats
         SET status = 'draining', last_heartbeat_at = NOW()
         WHERE worker_id = $1`,
        [this.workerId]
      );

      // Step 2: 等待当前任务完成（最多 30 秒）
      if (this.inFlightTasks.size > 0) {
        logger.info(`等待 ${this.inFlightTasks.size} 个任务完成...`);
        await this.waitForTasks(this.shutdownTimeoutMs);
      }

      // Step 3: 释放所有租约
      await pg.query(
        `DELETE FROM qd_analysis_leases WHERE worker_id = $1`,
        [this.workerId]
      );

      // Step 4: 关闭连接池
      await pg.end();
      await redis.quit();

      logger.info('优雅关闭完成');
    } catch (err) {
      logger.error('关闭过程中出错', { error: err.message });
    }

    process.exit(0);
  }

  private waitForTasks(timeoutMs: number): Promise<void> {
    return new Promise(resolve => {
      const check = setInterval(() => {
        if (this.inFlightTasks.size === 0) {
          clearInterval(check);
          resolve();
        }
      }, 500);

      setTimeout(() => {
        clearInterval(check);
        getLogger(this.workerId).warn(
          `超时等待 ${this.inFlightTasks.size} 个任务，强制退出`
        );
        resolve();
      }, timeoutMs);
    });
  }

  // 任务跟踪
  trackTask(taskId: string) { this.inFlightTasks.add(taskId); }
  completeTask(taskId: string) { this.inFlightTasks.delete(taskId); }
  canAcceptWork(): boolean { return !this.isShuttingDown; }
}
